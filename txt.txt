# ================ ЗАГРУЗКА/СОХРАНЕНИЕ ПОЛЬЗОВАТЕЛЕЙ ================
def load_users():
    """Загружает пользователей из файла"""
    global computer_users
    try:
        with open(USERS_FILE, 'r', encoding='utf-8') as f:
            computer_users = json.load(f)
    except FileNotFoundError:
        computer_users = {}
    except:
        computer_users = {}

def save_users():
    """Сохраняет пользователей в файл"""
    try:
        with open(USERS_FILE, 'w', encoding='utf-8') as f:
            json.dump(computer_users, f, ensure_ascii=False, indent=2)
    except:
        pass

# Загружаем пользователей при старте
load_users()

# ================ ФУНКЦИЯ ПОЛУЧЕНИЯ АКТИВНОГО ПОЛЬЗОВАТЕЛЯ ================
def get_active_user(pc_name):
    """
    Получает имя активного пользователя на компьютере через quser
    Возвращает: (username, session_id) или (None, None)
    """
    try:
        # Команда для получения активных сессий
        command = f'quser /server:{pc_name}'
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=3
        )
        
        if result.returncode != 0:
            return None, None
        
        # Парсим вывод quser
        lines = result.stdout.strip().split('\n')
        if len(lines) < 2:
            return None, None
        
        # Пропускаем заголовок
        for line in lines[1:]:
            parts = line.split()
            if len(parts) >= 2:
                username = parts[0]
                # Проверяем, что сессия активна (не Disconnected)
                if 'Disc' not in line and 'Выход' not in line:
                    return username, None
        
        return None, None
        
    except subprocess.TimeoutExpired:
        return None, None
    except Exception:
        return None, None

def update_all_users():
    """Обновляет информацию о пользователях на всех компьютерах"""
    global computer_users
    
    # Создаем список всех ПК
    all_pcs = [f"{COMPUTER_PREFIX}{i:02d}" for i in range(1, COMPUTER_COUNT + 1)]
    
    for pc in all_pcs:
        username, _ = get_active_user(pc)
        if username:
            computer_users[pc] = username
        else:
            # Если пользователь не найден, проверяем не залогинен ли он через другой метод
            # Или просто оставляем предыдущее значение с пометкой "неактивен"
            if pc in computer_users:
                # Проверяем, не устарела ли запись
                pass
    
    save_users()

def get_whoami(pc_name):
    """Альтернативный метод получения пользователя"""
    try:
        command = f'wmic /node:{pc_name} computersystem get username'
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=3
        )
        if result.returncode == 0:
            lines = result.stdout.strip().split('\n')
            for line in lines[1:]:
                line = line.strip()
                if line and '\\' in line:
                    return line.split('\\')[-1]
    except:
        pass
    return None

@app.route('/admin')
def admin_page():
    """Админ-панель"""
    # Обновляем пользователей
    update_all_users()
    
    online_count = len(computer_users)
    
    stats = {
        'total': len(call_log),
        'success': sum(1 for c in call_log if c['status'] == 'ok'),
        'failed': sum(1 for c in call_log if c['status'] == 'error'),
        'online': online_count
    }
    
    return render_template_string(
        ADMIN_PAGE,
        ip=LOCAL_IP,
        prefix=COMPUTER_PREFIX,
        users=computer_users,
        call_log=call_log,
        stats=stats,
        update_time=datetime.now().strftime('%H:%M:%S')
    )

@app.route('/api/call', methods=['POST'])
def make_call():
    """Вызов специалиста"""
    data = request.json
    pc_number = data.get('pc')
    message_text = data.get('message', '').strip()
    user_name = data.get('user', '')
    
    if not pc_number or pc_number < 1 or pc_number > COMPUTER_COUNT:
        return jsonify({'status': 'error', 'message': f'Неверный номер (1-{COMPUTER_COUNT})'})
    
    pc_name = f"{COMPUTER_PREFIX}{pc_number:02d}"
    return send_msg(pc_name, message_text, user_name)

def send_msg(pc_name, message_text, user_name=''):
    """Отправка сообщения через msg"""
    # Формируем сообщение
    if message_text and user_name:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nПользователь: {user_name}\nСообщение: {message_text}"
    elif message_text:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nСообщение: {message_text}"
    elif user_name:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nПользователь: {user_name}"
    else:
        full_message = "ВНИМАНИЕ! Подойдите в обработку"
    
    # Экранируем кавычки для cmd
    full_message = full_message.replace('"', '\\"')
    
    command = f'msg * /server:{pc_name} "{full_message}"'
    
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=5
        )
        
        if result.returncode == 0:
            success = True
            msg = f"Сообщение отправлено на {pc_name}"
        else:
            success = False
            error = result.stderr.strip() or 'Неизвестная ошибка'
            msg = f"Ошибка: {error}"
            
    except subprocess.TimeoutExpired:
        success = False
        msg = f"Таймаут: {pc_name} не отвечает"
    except Exception as e:
        success = False
        msg = f"Ошибка: {str(e)}"
    
    # Логируем вызов
    call_log.append({
        'time': datetime.now().strftime('%H:%M:%S'),
        'pc': pc_name,
        'user': user_name if user_name else None,
        'text': message_text if message_text else None,
        'status': 'ok' if success else 'error',
        'message': msg
    })
    
    return jsonify({'status': 'ok' if success else 'error', 'message': msg})
