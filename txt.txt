# ================ ЗАГРУЗКА/СОХРАНЕНИЕ ПОЛЬЗОВАТЕЛЕЙ ================
def load_users():
    global computer_users
    try:
        with open(USERS_FILE, 'r', encoding='utf-8') as f:
            computer_users = json.load(f)
    except FileNotFoundError:
        computer_users = {}
    except:
        computer_users = {}

def save_users():
    try:
        with open(USERS_FILE, 'w', encoding='utf-8') as f:
            json.dump(computer_users, f, ensure_ascii=False, indent=2)
    except:
        pass

load_users()

# ================ ФУНКЦИЯ ПОЛУЧЕНИЯ АКТИВНОГО ПОЛЬЗОВАТЕЛЯ ================
def get_active_user(pc_name):
    try:
        command = f'quser /server:{pc_name}'
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=3
        )
        
        if result.returncode != 0:
            return None, None
        
        lines = result.stdout.strip().split('\n')
        if len(lines) < 2:
            return None, None
        
        for line in lines[1:]:
            parts = line.split()
            if len(parts) >= 2:
                username = parts[0]
                if 'Disc' not in line and 'Выход' not in line:
                    return username, None
        
        return None, None
        
    except subprocess.TimeoutExpired:
        return None, None
    except Exception:
        return None, None

def update_all_users():
    global computer_users
    all_pcs = [f"{COMPUTER_PREFIX}{i:02d}" for i in range(1, COMPUTER_COUNT + 1)]
    
    for pc in all_pcs:
        username, _ = get_active_user(pc)
        if username:
            computer_users[pc] = username
    
    save_users()

            const empty = [];
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    if (grid[i][j] === 0) empty.push([i, j]);
                }
            }
            if (empty.length === 0) return false;
            const [r, c] = empty[Math.floor(Math.random() * empty.length)];
            grid[r][c] = Math.random() < 0.9 ? 2 : 4;
            return true;

        function renderGrid() {
            const gridEl = document.getElementById('grid');
            gridEl.innerHTML = '';
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    const cell = document.createElement('div');

                    cell.className = 'cell';

        function move(direction) {
            if (!gameRunning) return;
            
            let moved = false;
            const newGrid = grid.map(row => [...row]);
            
            if (direction === 'left') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i]);
                    newGrid[i] = result.row;
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'right') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i].slice().reverse());
                    newGrid[i] = result.row.reverse();
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'up') {
                for (let j = 0; j < 4; j++) {
                    const col = [grid[0][j], grid[1][j], grid[2][j], grid[3][j]];
                    const result = slideRow(col);
                    for (let i = 0; i < 4; i++) newGrid[i][j] = result.row[i];
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'down') {
                for (let j = 0; j < 4; j++) {
                    const col = [grid[3][j], grid[2][j], grid[1][j], grid[0][j]];
                    const result = slideRow(col);
                    for (let i = 0; i < 4; i++) newGrid[3 - i][j] = result.row[i];
                    score += result.points;

@app.route('/api/call', methods=['POST'])
def make_call():
    data = request.json
    pc_number = data.get('pc')
    message_text = data.get('message', '').strip()
    user_name = data.get('user', '')
    
    if not pc_number or pc_number < 1 or pc_number > COMPUTER_COUNT:
        return jsonify({'status': 'error', 'message': f'Неверный номер (1-{COMPUTER_COUNT})'})
    
    pc_name = f"{COMPUTER_PREFIX}{pc_number:02d}"
    return send_msg(pc_name, message_text, user_name)

def send_msg(pc_name, message_text, user_name=''):
    """Отправка сообщения через msg"""
    if message_text and user_name:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nПользователь: {user_name}\nСообщение: {message_text}"
    elif message_text:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nСообщение: {message_text}"
    elif user_name:
        full_message = f"ВНИМАНИЕ! Подойдите в обработку\n\nПользователь: {user_name}"
    else:
        full_message = "ВНИМАНИЕ! Подойдите в обработку"
    
    full_message = full_message.replace('"', '\\"')
    command = f'msg * /server:{pc_name} "{full_message}"'
    
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=5
        )
        
        if result.returncode == 0:
            success = True
            msg = f"Сообщение отправлено на {pc_name}"
        else:
            success = False
            error = result.stderr.strip() or 'Неизвестная ошибка'
            msg = f"Ошибка: {error}"
            
    except subprocess.TimeoutExpired:
        success = False
        msg = f"Таймаут: {pc_name} не отвечает"
    except Exception as e:
        success = False
        msg = f"Ошибка: {str(e)}"
    
    call_log.append({
        'time': datetime.now().strftime('%H:%M:%S'),
        'pc': pc_name,
        'user': user_name if user_name else None,
        'text': message_text if message_text else None,
        'status': 'ok' if success else 'error',
        'message': msg
    })
    
    return jsonify({'status': 'ok' if success else 'error', 'message': msg})
