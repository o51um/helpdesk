@echo off
echo УДАЛЕНИЕ ВСЕХ ФАЙЛОВ И ПАПОК ИЗ TEMP...

:: 1. Удаляем все файлы (включая скрытые)
del /f /s /q /a "%TEMP%\*.*" 2>nul

:: 2. Удаляем все папки (рекурсивно)
for /d %%i in ("%TEMP%\*") do (
    rmdir /s /q "%%i" 2>nul
)

:: 3. Создаём папку заново (чтобы не было ошибок)
mkdir "%TEMP%" 2>nul

echo ГОТОВО! Все файлы и папки удалены.
pause
