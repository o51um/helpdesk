def login_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not session.get('logged_in'):
            return redirect(url_for('login_page'))
        return f(*args, **kwargs)
    return decorated_function

# ================ РЕЙТИНГИ ================
def load_ranking():
    try:
        with open(RANKING_FILE, 'r', encoding='utf-8') as f:
            return json.load(f)
    except FileNotFoundError:
        return []
    except:
        return []

def save_ranking(ranking):
    try:
        with open(RANKING_FILE, 'w', encoding='utf-8') as f:
            json.dump(ranking, f, ensure_ascii=False, indent=2)
    except:
        pass

def add_score(name, score):
    ranking = load_ranking()
    ranking.append({
        'name': name,
        'score': score,
        'date': datetime.now().strftime('%d.%m.%Y %H:%M')
    })
    ranking.sort(key=lambda x: x['score'], reverse=True)
    ranking = ranking[:50]
    save_ranking(ranking)
    return ranking

            grid = Array(4).fill(null).map(() => Array(4).fill(0));

        function addRandomTile() {
            const empty = [];
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    if (grid[i][j] === 0) empty.push([i, j]);
                }
            }
            if (empty.length === 0) return false;
            const [r, c] = empty[Math.floor(Math.random() * empty.length)];
            grid[r][c] = Math.random() < 0.9 ? 2 : 4;
            return true;

        function renderGrid() {
            const gridEl = document.getElementById('grid');
            gridEl.innerHTML = '';
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    const cell = document.createElement('div');
                    const val = grid[i][j];
                    cell.className = 'cell';
                    if (val > 0) {
                        cell.textContent = val;

                            cell.classList.add('cell-' + val);

        function move(direction) {
            if (!gameRunning) return;

            const newGrid = grid.map(row => [...row]);
            
            if (direction === 'left') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i]);
                    newGrid[i] = result.row;
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'right') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i].slice().reverse());
                    newGrid[i] = result.row.reverse();
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'up') {
                for (let j = 0; j < 4; j++) {
                    const col = [grid[0][j], grid[1][j], grid[2][j], grid[3][j]];

                    const result = slideRow(col);
                    for (let i = 0; i < 4; i++) newGrid[3 - i][j] = result.row[i];

                    points += newRow[i];

            }
            
            for (let i = 0; i < 4; i++) {

@app.route('/api/chief/call', methods=['POST'])
def chief_call():
    data = request.json
    pcs = data.get('pcs', [])
    message_text = data.get('message', '').strip()
    
    if not pcs:
        return jsonify({'status': 'error', 'message': 'Не указаны получатели'})
    
    if not message_text:
        return jsonify({'status': 'error', 'message': 'Введите сообщение'})
    
    results = []
    success_count = 0
    
    for pc in pcs:
        result = send_msg_to_pc(pc, message_text)
        results.append({
            'pc': pc,
            'success': result['success'],
            'message': result['message']
        })
        if result['success']:
            success_count += 1
    
    chief_call_log.append({
        'time': datetime.now().strftime('%H:%M:%S'),
        'pc': f"Группа ({len(pcs)} шт.)",
        'text': message_text,
        'status': 'ok' if success_count > 0 else 'error',
        'message': f"Отправлено на {success_count}/{len(pcs)} окон"
    })
    
    return jsonify({
        'status': 'ok' if success_count > 0 else 'error',
        'message': f"Отправлено на {success_count}/{len(pcs)} окон",
        'results': results
    })

def send_msg_to_pc(pc_name, message_text):
    message_text = message_text.replace('\n', ' ').replace('\r', ' ')
    message_text = re.sub(r'\s+', ' ', message_text).strip()
    full_message = message_text.replace('"', '\\"')
    
    command = f'msg * /server:{pc_name} "{full_message}"'
    
    try:
        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=5
        )
        
        if result.returncode == 0:
            return {'success': True, 'message': f"Успешно отправлено на {pc_name}"}
        else:
            error = result.stderr.strip() or 'Неизвестная ошибка'
            return {'success': False, 'message': f"Ошибка: {error}"}
            
    except subprocess.TimeoutExpired:
        return {'success': False, 'message': f"Таймаут: {pc_name} не отвечает"}
    except Exception as e:
        return {'success': False, 'message': f"Ошибка: {str(e)}"}

def send_msg(pc_name, message_text, log, is_chief=False):
    result = send_msg_to_pc(pc_name, message_text)
    
    if is_chief:
        log_to = chief_call_log
    else:
        log_to = call_log
    
    log_to.append({
        'time': datetime.now().strftime('%H:%M:%S'),
        'pc': pc_name,
        'text': message_text if message_text else None,
        'status': 'ok' if result['success'] else 'error',
        'message': result['message']
    })
    
    return jsonify({'status': 'ok' if result['success'] else 'error', 'message': result['message']})
