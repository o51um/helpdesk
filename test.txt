    <script>
        let grid = [];
        let score = 0;
        let gameRunning = true;
        let finalScore = 0;
        let gameEnded = false;
        let rankingData = [];

        // ========== ИГРА ==========
        function initGame() {
            grid = Array(4).fill(null).map(() => Array(4).fill(0));
            score = 0;
            gameRunning = true;
            gameEnded = false;
            document.getElementById('gameOver').style.display = 'none';
            document.getElementById('saveModal').classList.remove('active');
            addRandomTile();
            addRandomTile();
            renderGrid();
            updateScore();
            loadRanking();
        }

        function addRandomTile() {
            const empty = [];
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    if (grid[i][j] === 0) empty.push([i, j]);
                }
            }
            if (empty.length === 0) return false;
            const [r, c] = empty[Math.floor(Math.random() * empty.length)];
            grid[r][c] = Math.random() < 0.9 ? 2 : 4;
            return true;
        }

        function renderGrid() {
            const gridEl = document.getElementById('grid');
            gridEl.innerHTML = '';
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    const cell = document.createElement('div');
                    const val = grid[i][j];
                    cell.className = 'cell';
                    if (val > 0) {
                        cell.textContent = val;
                        if (val <= 2048) {
                            cell.classList.add('cell-' + val);
                        } else {
                            cell.classList.add('cell-super');
                        }
                    }
                    gridEl.appendChild(cell);
                }
            }
        }

        function updateScore() {
            document.getElementById('score').textContent = score;
        }

        function move(direction) {
            if (!gameRunning) return;
            
            let moved = false;
            const newGrid = grid.map(row => [...row]);
            
            if (direction === 'left') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i]);
                    newGrid[i] = result.row;
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'right') {
                for (let i = 0; i < 4; i++) {
                    const result = slideRow(grid[i].slice().reverse());
                    newGrid[i] = result.row.reverse();
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'up') {
                for (let j = 0; j < 4; j++) {
                    const col = [grid[0][j], grid[1][j], grid[2][j], grid[3][j]];
                    const result = slideRow(col);
                    for (let i = 0; i < 4; i++) newGrid[i][j] = result.row[i];
                    score += result.points;
                    if (result.moved) moved = true;
                }
            } else if (direction === 'down') {
                for (let j = 0; j < 4; j++) {
                    const col = [grid[3][j], grid[2][j], grid[1][j], grid[0][j]];
                    const result = slideRow(col);
                    for (let i = 0; i < 4; i++) newGrid[3 - i][j] = result.row[i];
                    score += result.points;
                    if (result.moved) moved = true;
                }
            }
            
            if (!moved) return;
            
            grid = newGrid;
            updateScore();
            
            if (!addRandomTile()) {
                gameRunning = false;
                gameEnded = true;
                finalScore = score;
                document.getElementById('finalScore').textContent = score;
                document.getElementById('gameOver').style.display = 'flex';
            }
            
            renderGrid();
            
            for (let i = 0; i < 4; i++) {
                for (let j = 0; j < 4; j++) {
                    if (grid[i][j] === 2048) {
                        setTimeout(() => alert('🎉 Поздравляем! Вы набрали 2048!'), 300);
                        return;
                    }
                }
            }
        }

        function slideRow(row) {
            let newRow = row.filter(val => val !== 0);
            let points = 0;
            let moved = false;
            
            for (let i = 0; i < newRow.length - 1; i++) {
                if (newRow[i] === newRow[i + 1]) {
                    newRow[i] *= 2;
                    points += newRow[i];
                    newRow.splice(i + 1, 1);
                    moved = true;
                }
            }
            
            while (newRow.length < 4) {
                newRow.push(0);
            }
            
            for (let i = 0; i < 4; i++) {
                if (row[i] !== newRow[i]) moved = true;
            }
            
            return { row: newRow, points: points, moved: moved };
        }

        // ========== РЕЙТИНГ ==========
        function loadRanking() {
            fetch('/api/ranking')
                .then(response => response.json())
                .then(data => {
                    rankingData = data;
                    renderRanking();
                })
                .catch(() => {});
        }

        function renderRanking() {
            const container = document.getElementById('rankingList');
            if (!rankingData || rankingData.length === 0) {
                container.innerHTML = '<div class="rank-empty">🏆 Пока нет рекордов<br>Станьте первым!</div>';
                return;
            }
            
            let html = '';
            rankingData.forEach((item, index) => {
                let posClass = '';
                let posText = (index + 1);
                if (index === 0) { posClass = 'gold'; posText = '🥇'; }
                else if (index === 1) { posClass = 'silver'; posText = '🥈'; }
                else if (index === 2) { posClass = 'bronze'; posText = '🥉'; }
                
                html += `
                    <div class="rank-item">
                        <span class="rank-pos ${posClass}">${posText}</span>
                        <span class="rank-name">${escapeHtml(item.name)}</span>
                        <span class="rank-score">${item.score}</span>
                        <span class="rank-date">${item.date}</span>
                    </div>
                `;
            });
            container.innerHTML = html;
        }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        function clearRanking() {
            if (confirm('Удалить все рекорды?')) {
                fetch('/api/ranking/clear', {
                    method: 'POST'
                })
                .then(response => response.json())
                .then(data => {
                    if (data.status === 'ok') {
                        loadRanking();
                    }
                });
            }
        }

        // ========== СОХРАНЕНИЕ РЕЗУЛЬТАТА ==========
        function showSaveScore() {
            if (finalScore > 0) {
                document.getElementById('modalScore').textContent = finalScore;
                document.getElementById('saveModal').classList.add('active');
                document.getElementById('playerName').value = '';
                setTimeout(() => document.getElementById('playerName').focus(), 100);
            }
        }

        function saveScore() {
            const name = document.getElementById('playerName').value.trim() || 'Аноним';
            
            fetch('/api/game/save_score', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    name: name,
                    score: finalScore
                })
            })
            .then(response => response.json())
            .then(data => {
                if (data.status === 'ok') {
                    alert('✅ Результат сохранен!\nИгрок: ' + name + '\nОчки: ' + finalScore);
                    loadRanking();
                } else {
                    alert('❌ Ошибка сохранения');
                }
                closeModalAndContinue();
            })
            .catch(() => {
                alert('❌ Ошибка соединения');
                closeModalAndContinue();
            });
        }

        function closeModalAndContinue() {
            document.getElementById('saveModal').classList.remove('active');
            initGame();
        }

        // ========== СОБЫТИЯ ==========
        document.getElementById('playerName').addEventListener('keydown', function(e) {
            if (e.key === 'Enter') {
                saveScore();
            }
        });

        document.addEventListener('keydown', (e) => {
            if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.key)) {
                e.preventDefault();
                const map = {
                    'ArrowUp': 'up',
                    'ArrowDown': 'down',
                    'ArrowLeft': 'left',
                    'ArrowRight': 'right'
                };
                move(map[e.key]);
            }
        });

        let startX = 0, startY = 0;
        document.addEventListener('touchstart', (e) => {
            startX = e.touches[0].clientX;
            startY = e.touches[0].clientY;
        });
        document.addEventListener('touchmove', (e) => {
            e.preventDefault();
        });
        document.addEventListener('touchend', (e) => {
            if (startX === 0 && startY === 0) return;
            const dx = e.changedTouches[0].clientX - startX;
            const dy = e.changedTouches[0].clientY - startY;
            if (Math.abs(dx) < 20 && Math.abs(dy) < 20) return;
            if (Math.abs(dx) > Math.abs(dy)) {
                move(dx > 0 ? 'right' : 'left');
            } else {
                move(dy > 0 ? 'down' : 'up');
            }
            startX = 0;
            startY = 0;
        });

        // Запуск
        initGame();
    </script>
</body>
</html>
